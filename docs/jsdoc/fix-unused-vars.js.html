import fs from 'fs';
import { createRequire } from 'module';

const require = createRequire(import.meta.url);
const glob = require('glob');

/**
 * Fixes unused variables by prefixing them with an underscore
 * @param {string} content - The file content
 * @returns {string} - The fixed content
 */
function fixUnusedVariables(content) {
  // Get all eslint-disable and eslint-disable-next-line comments to avoid modifying variables
  // that are explicitly disabled
  const disabledLines = [];
  const disableRegex = /\/\/ eslint-disable-next-line.*no-unused-vars|\/\* eslint-disable.*no-unused-vars.*\*\//g;
  let match;
  let lineNumber = 1;
  
  for (const line of content.split('\n')) {
    if (line.match(disableRegex)) {
      disabledLines.push(lineNumber);
      disabledLines.push(lineNumber + 1); // Also disable the next line
    }
    lineNumber++;
  }

  // Find all variable declarations
  const unusedVarsRegex = /(const|let|var|function|class)\s+([a-zA-Z][a-zA-Z0-9]*)\s*(?::|=|\(|{|extends)/g;
  
  const potentialUnusedVars = [];
  while ((match = unusedVarsRegex.exec(content)) !== null) {
    const varName = match[2];
    if (!varName.startsWith('_') && varName !== 'undefined' && varName !== 'null') {
      potentialUnusedVars.push(varName);
    }
  }

  // For each potential unused var, check if it's used elsewhere in the content
  let updatedContent = content;
  for (const varName of potentialUnusedVars) {
    // Skip common React hooks and built-ins
    if (['useState', 'useEffect', 'useRef', 'useMemo', 'useCallback', 'useContext', 'useReducer', 'use', 'require', 'module', 'exports'].includes(varName)) {
      continue;
    }
    
    // Count how many times the variable appears in the content outside of its declaration
    const declarations = new RegExp(`(const|let|var|function|class)\\s+${varName}\\s*(?::|=|\\(|{|extends)`, 'g');
    const declarationCount = (content.match(declarations) || []).length;
    
    // Count all usages of this variable name (including in strings, comments, etc.)
    const usages = new RegExp(`\\b${varName}\\b`, 'g');
    const usageCount = (content.match(usages) || []).length;
    
    // If the variable is only used in its declaration, it's likely unused
    if (usageCount <= declarationCount) {
      // Replace variable name with _varName
      updatedContent = updatedContent.replace(
        new RegExp(`(const|let|var|function|class)\\s+${varName}(\\s*(?::|=|\\(|{|extends))`, 'g'),
        `$1 _${varName}$2`
      );
    }
  }

  return updatedContent;
}

/**
 * Finds and processes all TypeScript files in the src directory
 */
async function processFiles() {
  // Find all TypeScript files in src directory
  const files = glob.sync('src/**/*.{ts,tsx}', {
    ignore: ['**/node_modules/**', '**/*.d.ts']
  });

  console.log(`Found ${files.length} TypeScript files to process`);
  let fixedFilesCount = 0;

  for (const file of files) {
    try {
      // Skip declaration files
      if (file.endsWith('.d.ts')) {
        console.log(`Skipping declaration file: ${file}`);
        continue;
      }

      const content = fs.readFileSync(file, 'utf8');
      const updatedContent = fixUnusedVariables(content);

      // Only write if content changed
      if (content !== updatedContent) {
        fs.writeFileSync(file, updatedContent);
        console.log(`✓ Updated ${file}`);
        fixedFilesCount++;
      }
    } catch (error) {
      console.error(`Error processing ${file}: ${error.message}`);
    }
  }

  console.log(`Unused variable fixing completed: ${fixedFilesCount} files updated`);
}

// Execute the script
processFiles().catch(error => {
  console.error('Error:', error);
  process.exit(1);
});
