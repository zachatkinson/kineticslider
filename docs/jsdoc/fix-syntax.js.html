import { createRequire } from 'module';

const require = createRequire(import.meta.url);
const glob = require('glob');

/**
 * Fixes common syntax errors in TypeScript files
 * @param {string} content - The file content
 * @returns {string} - The fixed content
 */
function fixSyntaxErrors(content) {
  let updatedContent = content;

  // Fix missing commas in argument lists and object literals
  updatedContent = updatedContent.replace(/(\w+)\s+(\w+):\s*([^,;\s{}()]+)\s*\)/g, '$1, $2: $3)');
  updatedContent = updatedContent.replace(/(\w+)\s+(\w+):\s*([^,;\s{}()]+)(\s*\n\s*[})])/g, '$1, $2: $3$4');
  
  // Fix missing semicolons at the end of type declarations
  updatedContent = updatedContent.replace(/(\w+):\s*([^;,\s{()]+)\s*$/gm, '$1: $2;');
  
  // Fix missing expressions
  updatedContent = updatedContent.replace(/=>\s*{(\s*)}|=>\s*$/g, '=> { return undefined; }');
  
  // Fix broken arrow functions without return types
  updatedContent = updatedContent.replace(/(\w+)\s*=>\s*{([^}]*)}/g, (match, param, body) => {
    if (!body.includes('return')) {
      return `${param} => { return ${body.trim() || 'undefined'}; }`;
    }
    return match;
  });

  // Fix missing colons in object type definitions
  updatedContent = updatedContent.replace(/(\w+)\s+(\w+)\s*([,}])/g, '$1: $2$3');
  
  // Fix missing parentheses
  updatedContent = updatedContent.replace(/([\w.]+)\s*\(\s*([^()]*[^,\s])(\s*{)/g, '$1($2)$3');
  
  // Fix expressions expected (incomplete expressions)
  updatedContent = updatedContent.replace(/(\w+)\s*=\s*$/gm, '$1 = undefined;');
  updatedContent = updatedContent.replace(/:\s*\[\s*$/gm, ': [];');
  updatedContent = updatedContent.replace(/:\s*{\s*$/gm, ': {};');
  
  // Fix type expected errors
  updatedContent = updatedContent.replace(/@type\s*$/gm, '@type {unknown}');
  
  // Fix missing arrow function parameters
  updatedContent = updatedContent.replace(/=>\s+\{/g, '() => {');
  
  // Fix element access without argument
  updatedContent = updatedContent.replace(/(\w+)\[\]([^\[])/g, '$1[0]$2');
  
  // Fix missing jsx closing brackets
  updatedContent = updatedContent.replace(/(<\w+[^>]*)\s*$/gm, '$1 />');
  
  return updatedContent;
}

/**
 * Finds and processes all TypeScript files in the src directory
 */
async function processFiles() {
  // Find all TypeScript files in src directory
  const files = glob.sync('src/**/*.{ts,tsx}', {
    ignore: ['**/node_modules/**', '**/*.d.ts']
  });

  console.log(`Found ${files.length} TypeScript files to process`);
  let fixedFilesCount = 0;

  for (const file of files) {
    try {
      // Skip declaration files
      if (file.endsWith('.d.ts')) {
        console.log(`Skipping declaration file: ${file}`);
        continue;
      }

      const content = fs.readFileSync(file, 'utf8');
      const updatedContent = fixSyntaxErrors(content);

      // Only write if content changed
      if (content !== updatedContent) {
        fs.writeFileSync(file, updatedContent);
        console.log(`✓ Updated ${file}`);
        fixedFilesCount++;
      }
    } catch (error) {
      console.error(`Error processing ${file}: ${error.message}`);
    }
  }

  console.log(`Syntax fixing completed: ${fixedFilesCount} files updated`);
}

// Execute the script
processFiles().catch(error => {
  console.error('Error:', error);
  process.exit(1);
});
