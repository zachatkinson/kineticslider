import fs from 'fs';
import { createRequire } from 'module';

const require = createRequire(import.meta.url);
const glob = require('glob');

/**
 * Fixes missing JSDoc @returns declarations
 * @param {string} content - The file content
 * @returns {string} - The fixed content
 */
function fixMissingReturns(content) {
  // Regex to find JSDoc blocks that don't have @returns tag but should
  const jsdocBlocksRegex = /\/\*\*\s*\n(?:\s*\*[^\n]*\n)*?\s*\*\/\s*\n\s*(export\s+)?(async\s+)?(function\s+\w+|\w+\s*=\s*(?:async\s+)?(?:function|\([^)]*\)\s*=>))/g;
  
  // Improve by checking for the presence of @returns or @return tag
  const returnsTagRegex = /@returns?/;
  
  // Process each JSDoc block
  let updatedContent = content;
  let match;
  while ((match = jsdocBlocksRegex.exec(content)) !== null) {
    const fullMatch = match[0];
    const blockEndIndex = fullMatch.lastIndexOf('*/');
    
    // Check if the block already has a @returns tag
    if (!returnsTagRegex.test(fullMatch.substr(0, blockEndIndex))) {
      // Get the function signature to determine return type
      const functionSignature = fullMatch.substring(blockEndIndex + 2).trim();
      let returnType = '{unknown}';
      
      // Try to extract return type from function signature
      const returnTypeMatch = functionSignature.match(/:\s*([^{=;]+)(?:\s*(?:{|=>|\(|\n|$))/);
      if (returnTypeMatch) {
        returnType = `{${returnTypeMatch[1].trim()}}`;
      } else if (functionSignature.includes('Promise<')) {
        const promiseMatch = functionSignature.match(/Promise<([^>]+)>/);
        if (promiseMatch) {
          returnType = `{Promise<${promiseMatch[1]}>}`;
        }
      } else if (functionSignature.includes('=> void')) {
        returnType = '{void}';
      } else if (functionSignature.includes('=> boolean')) {
        returnType = '{boolean}';
      } else if (functionSignature.includes('=> string')) {
        returnType = '{string}';
      } else if (functionSignature.includes('=> number')) {
        returnType = '{number}';
      }
      
      // Add @returns tag before the closing */
      const replacement = fullMatch.slice(0, blockEndIndex) + 
                         ` * @returns ${returnType} - The return value\n */` +
                         fullMatch.slice(blockEndIndex + 2);
      
      updatedContent = updatedContent.replace(fullMatch, replacement);
    }
  }
  
  return updatedContent;
}

/**
 * Finds and processes all TypeScript files in the src directory
 */
async function processFiles() {
  // Find all TypeScript files in src directory
  const files = glob.sync('src/**/*.{ts,tsx}', {
    ignore: ['**/node_modules/**', '**/*.d.ts']
  });

  console.log(`Found ${files.length} TypeScript files to process`);
  let fixedFilesCount = 0;

  for (const file of files) {
    try {
      // Skip declaration files
      if (file.endsWith('.d.ts')) {
        console.log(`Skipping declaration file: ${file}`);
        continue;
      }

      const content = fs.readFileSync(file, 'utf8');
      const updatedContent = fixMissingReturns(content);

      // Only write if content changed
      if (content !== updatedContent) {
        fs.writeFileSync(file, updatedContent);
        console.log(`✓ Updated ${file}`);
        fixedFilesCount++;
      }
    } catch (error) {
      console.error(`Error processing ${file}: ${error.message}`);
    }
  }

  console.log(`JSDoc @returns fixing completed: ${fixedFilesCount} files updated`);
}

// Execute the script
processFiles().catch(error => {
  console.error('Error:', error);
  process.exit(1);
});
