#!/usr/bin/env node

import { removeCommentsFromTypeScriptFiles } from '../utils/removeComments';

/**
 * Main function to run the comment cleaning process
 */
async function main() {
  // Default directories to process
  const defaultDirectories = ['migration-tools'];
  
  // Get directories from command line arguments or use defaults
  const directories = process.argv.slice(2).length > 0 
    ? process.argv.slice(2) 
    : defaultDirectories;
  
  console.log(`Cleaning comments from directories: ${directories.join(', ')}`);
  
  try {
    // Process all TypeScript files in the specified directories
    const results = await removeCommentsFromTypeScriptFiles(directories);
    
    // Log results
    console.log('===== RESULTS =====');
    console.log(`Successfully processed: ${results.successful} files`);
    console.log(`Failed to process: ${results.failed} files`);
  } catch (error) {
    console.error('Error running clean comments script:', error);
    process.exit(1);
  }
}

// Execute the main function
main();
