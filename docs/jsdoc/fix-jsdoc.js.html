#!/usr/bin/env node

/**
 * Script to automatically fix common JSDoc issues in the codebase
 * @module scripts/fix-jsdoc
 */

import fs from 'fs';
import path from 'path';
import { glob } from 'glob';

/**
 * Fixes missing JSDoc returns declarations
 * @param {string} content - The file content to fix
 * @returns {string} The fixed content
 */
function fixMissingReturns(content) {
  // Match function declarations with JSDoc comments that are missing @returns
  const regex = /\/\*\*[\s\S]*?\*\/\s*(export\s+)?(async\s+)?function\s+\w+\s*\([^)]*\)\s*:\s*\w+/g;
  return content.replace(regex, (match) => {
    if (!match.includes('@returns')) {
      // Add @returns tag before the closing */
      return match.replace(/\*\//, '* @returns {ReturnType} The return value\n */');
    }
    return match;
  });
}

/**
 * Fixes missing JSDoc descriptions
 * @param {string} content - The file content to fix
 * @returns {string} The fixed content
 */
function fixMissingDescriptions(content) {
  // Match JSDoc comments that don't have a description after the opening /**
  const regex = /\/\*\*\s*\n\s*\* @/g;
  return content.replace(regex, '/**\n * Description\n * @');
}

/**
 * Fixes invalid JSDoc tag names
 * @param {string} content - The file content to fix
 * @returns {string} The fixed content
 */
function fixInvalidTags(content) {
  // First, fix any duplicated "-notes-notes" patterns with a single "-notes"
  content = content.replace(/@(component|performance|security|accessibility|error|state|style|validation|compatibility|internal)-notes-notes(-related)?/g, '@$1-notes$2');

  // Then handle any other patterns
  const tagMappings = {
    '@component': '@component-notes',
    '@performance': '@performance-notes',
    '@security': '@security-notes',
    '@accessibility': '@accessibility-notes',
    '@error': '@error-notes',
    '@state': '@state-notes',
    '@style': '@style-notes',
    '@validation': '@validation-notes',
    '@compatibility': '@compatibility-notes',
    '@internal': '@internal-notes',
  };

  // Replace all occurrences of invalid tags with their corrected versions
  Object.entries(tagMappings).forEach(([invalidTag, correctedTag]) => {
    // Match the tag alone or with -related suffix
    const tagRegex = new RegExp(`${invalidTag}(-related)?`, 'g');
    content = content.replace(tagRegex, `${correctedTag}$1`);
  });

  return content;
}

/**
 * Fixes missing JSDoc example descriptions
 * @param {string} content - The file content to fix
 * @returns {string} The fixed content
 */
function fixMissingExampleDescriptions(content) {
  // Find @example tags without descriptions
  const regex = /(@example\s*\n)/g;
  return content.replace(regex, '@example Example usage\n');
}

/**
 * Fixes @event tags that need a name parameter
 * @param {string} content - File content to process
 * @returns {string} - Updated content with fixed event tags
 */
function fixEventTags(content) {
  // Find @event tags without parameters and add a default name
  const eventTagRegex = /(@event)(?!\s+[\w\.]+)/g;
  return content.replace(eventTagRegex, '$1 onChange');
}

/**
 * Adds missing description to JSDoc blocks that don't have one
 * @param {string} content - File content to process
 * @returns {string} - Content with descriptions added to JSDoc blocks
 */
function fixMissingDescription(content) {
  // Find JSDoc blocks that don't have a description (first line after /** is a tag)
  const regex = /(\/\*\*\s*\n\s*\*)(\s*@)/g;
  return content.replace(regex, '$1 This is a description for this JSDoc block.\n *$2');
}

/**
 * Fix JSDoc issues in the given content
 * @param {string} content - File content to process
 * @returns {string} - Fixed content
 */
function fixJSDocIssues(content) {
  let fixedContent = content;
  fixedContent = fixMissingReturns(fixedContent);
  fixedContent = fixInvalidTags(fixedContent);
  fixedContent = fixEventTags(fixedContent);
  fixedContent = fixMissingDescription(fixedContent);
  fixedContent = fixMissingExampleDescriptions(fixedContent);
  return fixedContent;
}

/**
 * Process a file to fix JSDoc issues
 * @param {string} filePath - Path to the file
 * @returns {Promise<void>}
 */
async function processFile(filePath) {
  try {
    console.log(`Processing ${filePath}`);
    
    // Read the file
    let content = fs.readFileSync(filePath, 'utf8');
    
    // Apply fixes
    content = fixJSDocIssues(content);
    
    // Write the updated content back to the file
    fs.writeFileSync(filePath, content, 'utf8');
    
    console.log(`✓ Updated ${filePath}`);
  } catch (error) {
    console.error(`Error processing ${filePath}:`, error);
  }
}

/**
 * Main function to run the script
 * @returns {Promise<void>}
 */
async function main() {
  // Define the patterns to match
  const patterns = [
    'src/**/*.ts',
    'src/**/*.tsx',
    '!src/**/*.d.ts',
    '!src/node_modules/**',
    '!src/**/*.test.ts',
    '!src/**/*.test.tsx',
    '!src/**/__tests__/**'
  ];

  // Get all files matching the pattern
  const files = glob.sync(patterns, {
    cwd: process.cwd(),
    absolute: true,
  });

  console.log(`Found ${files.length} files to process`);

  // Process each file
  for (const file of files) {
    await processFile(file);
  }

  console.log('JSDoc fixing completed!');
}

main().catch(console.error);
